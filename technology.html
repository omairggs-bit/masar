/// Volumetric weight per air-freight industry standard: (L*W*H in cm) / 5000
#[no_mangle]
pub extern "C" fn volumetric_weight(l: f64, w: f64, h: f64) -> f64 {
    if l <= 0.0 || w <= 0.0 || h <= 0.0 { return 0.0; }
    (l * w * h) / 5000.0
}

#[no_mangle]
pub extern "C" fn max_f64(a: f64, b: f64) -> f64 {
    if a > b { a } else { b }
}

/// Core tiered-rate calculation shared by all shipment categories.
/// Returns the shipping-only subtotal (before flat add-ons).
#[no_mangle]
pub extern "C" fn tiered_price(
    base: f64,
    included_kg: f64,
    per_kg: f64,
    billable_weight: f64,
    cold_fee: f64,
    express_fee: f64,
) -> f64 {
    let extra = if billable_weight > included_kg { billable_weight - included_kg } else { 0.0 };
    base + extra * per_kg + cold_fee + express_fee
}

#[no_mangle]
pub extern "C" fn insurance_fee(declared_value: f64, rate: f64, min_fee: f64) -> f64 {
    if declared_value <= 0.0 { return 0.0; }
    let v = declared_value * rate;
    max_f64(v, min_fee)
}

/// Rounds to 2 decimal places without std (manual banker-free rounding).
#[no_mangle]
pub extern "C" fn round2(n: f64) -> f64 {
    let scaled = n * 100.0;
    let floor = (scaled as i64) as f64;
    let diff = scaled - floor;
    let rounded = if diff >= 0.5 { floor + 1.0 } else { floor };
    rounded / 100.0
}

#[no_mangle]
pub extern "C" fn direct_price(shipping_subtotal: f64, direct_factor: f64) -> f64 {
    round2(shipping_subtotal * direct_factor)
}

#[no_mangle]
pub extern "C" fn savings_amount(direct_total: f64, our_total: f64) -> f64 {
    let d = direct_total - our_total;
    if d > 0.0 { round2(d) } else { 0.0 }
}

// Masar tracking resolver — freestanding C++ compiled directly to WebAssembly
// (no libc, no emscripten): a deterministic polynomial hash turns any
// tracking code into a consistent pseudo-shipment state.
extern "C" {

// Static scratch buffer JS writes the tracking code's UTF-16 units into
// before calling masar_hash. No allocator needed since this is a fixed,
// generously-sized buffer (freestanding module, no libc/malloc).
static unsigned short g_buffer[256];

unsigned short* masar_buffer_ptr() {
    return g_buffer;
}

// 32-bit FNV-1a-style polynomial hash over the raw UTF-16 code units
// passed in from JS (avoids needing any string/libc support here).
unsigned int masar_hash(int len) {
    unsigned int h = 2166136261u;
    for (int i = 0; i < len && i < 256; i++) {
        h ^= (unsigned int)g_buffer[i];
        h *= 16777619u;
    }
    return h;
}

// Which of the 6 timeline stages (0..5) this tracking code is currently at.
int masar_stage(unsigned int hash) {
    return (int)(hash % 6u);
}

// Index into the JS-side carrier list.
int masar_carrier_index(unsigned int hash, int carrierCount) {
    if (carrierCount <= 0) return 0;
    return (int)(hash % (unsigned int)carrierCount);
}

// Index into the JS-side destination list.
int masar_dest_index(unsigned int hash, int destCount) {
    if (destCount <= 0) return 0;
    return (int)((hash / 7u) % (unsigned int)destCount);
}

// Days from "today" until estimated delivery, factoring in current stage
// (further along the timeline = fewer days remaining).
int masar_eta_days(unsigned int hash, int stage) {
    int remaining = 5 - stage;
    if (remaining < 0) remaining = 0;
    int jitter = (int)(hash % 3u);
    return remaining + jitter;
}

} // extern "C"
